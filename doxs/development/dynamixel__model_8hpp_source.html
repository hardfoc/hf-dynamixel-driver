157inline bool ModeSupported(const ModelDescriptor& m, OperatingMode mode) noexcept {
158 switch (mode) {
159 case OperatingMode::Velocity: return m.supports_velocity;
160 case OperatingMode::Position: return m.supports_position;
161 case OperatingMode::ExtendedPosition: return m.supports_extended_position;
162 case OperatingMode::Pwm: return m.supports_pwm;
163 }
164 return false;
165}

167inline const ModelDescriptor* FindModel(uint16_t model_number) noexcept {
168 for (const ModelDescriptor* m : kKnownModels) {
169 if (m->model_number == model_number) {
170 return m;
171 }
172 }
173 return nullptr;
