101 int writePort(uint8_t* packet, int length) override {
102 if (packet == nullptr || length <= 0) {
103 return 0;
104 }
105 last_echo_ok_ = true;
106 std::size_t written = 0;
107 while (written < static_cast<std::size_t>(length)) {
108 const TransportIo io = transport_.write_some(
109 packet + written, static_cast<std::size_t>(length) - written);
110 if (io.error != DriverError::None) {
111 return -1;
112 }
113 if (io.accepted == 0) {
114 break;
115 }
116 written += io.accepted;
117 }
118 if (written != static_cast<std::size_t>(length)) {
119 return static_cast<int>(written);
120 }
121 if (transport_.finish_transmit() != DriverError::None) {
122 return -1;
123 }
124 if (config_.echo_policy == EchoPolicy::ExpectedLocalEcho) {
125 if (!ConsumeEcho(packet, static_cast<std::size_t>(length))) {
126 last_echo_ok_ = false;
127 return -1;
128 }
129 }
130 return static_cast<int>(written);

169 bool ConsumeEcho(const uint8_t* expected, std::size_t length) noexcept {
170 uint8_t scratch[64];
171 std::size_t matched = 0;
172 const uint64_t deadline = transport_.now_us() + 20 * 1000;
173 while (matched < length) {
174 const std::size_t chunk = std::min(length - matched, sizeof(scratch));
175 const TransportIo io = transport_.read_some(scratch, chunk);
176 if (io.error == DriverError::SerialError) {
177 return false;
178 }
179 if (io.accepted == 0) {
180 if (transport_.now_us() >= deadline) {
181 return false;
182 }
183 transport_.yield();
184 continue;
185 }
186 for (std::size_t i = 0; i < io.accepted; ++i) {
187 if (scratch[i] != expected[matched + i]) {
188 return false;
189 }
190 }
191 matched += io.accepted;
192 }
193 return true;
194 }
