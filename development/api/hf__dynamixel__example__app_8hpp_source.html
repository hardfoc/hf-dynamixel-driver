47inline bool OpenExampleBus(dynamixel::Bus& bus, const char* tag) noexcept {
48 const auto opened = bus.Open(ExampleTransportConfig());
49 if (!opened.ok()) {
50 ESP_LOGE(tag, "bus open failed: %s comm=%d",
51 dynamixel::ToString(opened.error).data(), opened.detail.comm_code);
52 return false;
53 }
54 ESP_LOGI(tag, "uart%u tx=%d rx=%d dir=%d baud=%u echo=%s",
55 static_cast<unsigned>(CONFIG_HF_DYNAMIXEL_UART_PORT),
56 CONFIG_HF_DYNAMIXEL_UART_TX_GPIO, CONFIG_HF_DYNAMIXEL_UART_RX_GPIO,
57 CONFIG_HF_DYNAMIXEL_UART_DIR_GPIO, CONFIG_HF_DYNAMIXEL_BAUD,
58 (ExampleTransportConfig().echo_policy == dynamixel::EchoPolicy::ExpectedLocalEcho)
59 ? "local"

67inline dynamixel::DriverResult<dynamixel::PingInfo> IdentifyOrBind(
68 dynamixel::Device& dev, const char* tag) noexcept {
69 auto ping = dev.Identify();
70 if (ping.ok()) {
71 ESP_LOGI(tag, "id=%u model=%u (%s) fw=%u", ping.value.id,
72 ping.value.model_number,
73 (dev.model() != nullptr) ? dev.model()->name : "?",
74 ping.value.firmware_version);
75 return ping;
76 }
77 if (ping.error == dynamixel::DriverError::UnsupportedRequest &&
78 ping.value.model_number != 0) {
79 ESP_LOGW(tag, "id=%u model=%u is not in the catalog", ping.value.id,
80 ping.value.model_number);
81#if defined(CONFIG_HF_DYNAMIXEL_BIND_XSERIES_FALLBACK) && \
82 CONFIG_HF_DYNAMIXEL_BIND_XSERIES_FALLBACK
83 dev.BindXSeriesFallback();
84 ESP_LOGW(tag, "bound X-series fallback (shared control table)");
85 ping.error = dynamixel::DriverError::None;
86 return ping;
87#endif
88 }
89 ESP_LOGE(tag, "identify failed: %s comm=%d",
90 dynamixel::ToString(ping.error).data(), ping.detail.comm_code);
91 return ping;
92}
